with public_tables as (
  select c.relname as table_name
  from pg_class c join pg_namespace n on n.oid=c.relnamespace
  where n.nspname='public' and c.relkind='r'
),
admin_policy_coverage as (
  select t.table_name,
    bool_or(e.cmd='SELECT') as has_select,
    bool_or(e.cmd='INSERT') as has_insert,
    bool_or(e.cmd='UPDATE') as has_update,
    bool_or(e.cmd='DELETE') as has_delete
  from public_tables t
  left join (
    select tablename, unnest(case when cmd='ALL' then array['SELECT','INSERT','UPDATE','DELETE'] else array[cmd] end) as cmd
    from pg_policies
    where schemaname='public'
      and (coalesce(qual,'') ilike '%is_admin(auth.uid())%' or coalesce(with_check,'') ilike '%is_admin(auth.uid())%')
  ) e on e.tablename=t.table_name
  group by t.table_name
),
checks as (
  select 'RLS enabled on all public base tables'::text as requirement,
         case when (select count(*) from pg_class c join pg_namespace n on n.oid=c.relnamespace where n.nspname='public' and c.relkind='r' and not c.relrowsecurity)=0 then 'PASS' else 'FAIL' end as status,
         (select count(*)::text from pg_class c join pg_namespace n on n.oid=c.relnamespace where n.nspname='public' and c.relkind='r' and not c.relrowsecurity) as detail
  union all
  select 'Admin CRUD policies on all public tables',
         case when (select count(*) from admin_policy_coverage where not (coalesce(has_select,false) and coalesce(has_insert,false) and coalesce(has_update,false) and coalesce(has_delete,false)))=0 then 'PASS' else 'FAIL' end,
         (select coalesce(string_agg(table_name, ', '), '0') from admin_policy_coverage where not (coalesce(has_select,false) and coalesce(has_insert,false) and coalesce(has_update,false) and coalesce(has_delete,false)))
  union all
  select 'is_admin(uid uuid) is SECURITY DEFINER',
         case when exists(select 1 from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and p.proname='is_admin' and pg_get_function_identity_arguments(p.oid)='uid uuid' and p.prosecdef) then 'PASS' else 'FAIL' end,
         (select coalesce(string_agg(pg_get_userbyid(p.proowner), ', '), 'missing') from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and p.proname='is_admin' and pg_get_function_identity_arguments(p.oid)='uid uuid')
  union all
  select 'staff_* and students_* policies include auth_user_id IS NOT NULL',
         case when (select count(*) from pg_policies where schemaname='public' and tablename in ('staff','students') and policyname like '%_own' and coalesce(qual,'') not ilike '%auth_user_id IS NOT NULL%' and coalesce(with_check,'') not ilike '%auth_user_id IS NOT NULL%')=0 then 'PASS' else 'FAIL' end,
         (select count(*)::text from pg_policies where schemaname='public' and tablename in ('staff','students') and policyname like '%_own' and coalesce(qual,'') not ilike '%auth_user_id IS NOT NULL%' and coalesce(with_check,'') not ilike '%auth_user_id IS NOT NULL%')
  union all
  select 'NOT NULL: students.auth_user_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='students' and column_name='auth_user_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='students' and column_name='auth_user_id')
  union all
  select 'NOT NULL: staff.auth_user_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='staff' and column_name='auth_user_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='staff' and column_name='auth_user_id')
  union all
  select 'NOT NULL: student_course_history.student_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='student_course_history' and column_name='student_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='student_course_history' and column_name='student_id')
  union all
  select 'NOT NULL: student_course_history.course_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='student_course_history' and column_name='course_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='student_course_history' and column_name='course_id')
  union all
  select 'NOT NULL: program_requirement_blocks.program_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='program_requirement_blocks' and column_name='program_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='program_requirement_blocks' and column_name='program_id')
  union all
  select 'NOT NULL: program_requirement_courses.block_id', case when (select is_nullable='NO' from information_schema.columns where table_schema='public' and table_name='program_requirement_courses' and column_name='block_id') then 'PASS' else 'FAIL' end, (select is_nullable from information_schema.columns where table_schema='public' and table_name='program_requirement_courses' and column_name='block_id')

  union all
